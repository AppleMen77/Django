# ═══════════════════════════════════════════════
#           🐧 DJANGO QUICK START
# ═══════════════════════════════════════════════

# 1. Создать виртуальное окружение
python -m venv .venv

# 2. Активировать его
source .venv/bin/activate

# 3. Установить Django
pip install Django

# 4. Посмотреть все команды
django-admin

# 5. Создать проект
django-admin startproject BlogSite

# 6. Перейти в папку проекта
cd BlogSite

# 7. Запустить сервер
python manage.py runserver

# ═══════════════════════════════════════════════
# Сайт доступен по адресу: http://127.0.0.1:8000
# Остановить сервер: Ctrl + C
# ═══════════════════════════════════════════════

# 8. Создать приложение
python manage.py startapp main

# ═══════════════════════════════════════════════
#        НЕ ЗАБУДЬ ЗАРЕГИСТРИРОВАТЬ ПРИЛОЖЕНИЕ
# ═══════════════════════════════════════════════
# Добавь 'main' в INSTALLED_APPS в settings.py:
#
# INSTALLED_APPS = [
#     ...
#     'main',
# ]
# ═══════════════════════════════════════════════

# 9. Применить миграции (база данных)
python manage.py migrate

# ═══════════════════════════════════════════════
# Миграции создают таблицы для:
#   • сессий
#   • пользователей
#   • прав доступа
#   • и других встроенных моделей
# ═══════════════════════════════════════════════

# 10. Создать суперпользователя (админа)
python manage.py createsuperuser

# ═══════════════════════════════════════════════
# Терминал запросит:
#   Имя пользователя: admin
#   Email: admin@example.com
#   Пароль: ******** (не видно в терминале)
#   Пароль (ещё раз): ********
# ═══════════════════════════════════════════════
# Админка доступна по адресу:
# http://127.0.0.1:8000/admin/
# ═══════════════════════════════════════════════

# 12. Создать файлы миграций (для своих моделей)
python manage.py makemigrations

# ═══════════════════════════════════════════════
# Выполняется, когда ты:
#   • создал новые модели в models.py
#   • изменил существующие модели
#   • добавил или удалил поля
#
# После makemigrations снова запусти migrate:
# python manage.py migrate
# ═══════════════════════════════════════════════
